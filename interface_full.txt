import os
import sys
import math
import random
import threading
import tkinter as tk

try:
    import pyttsx3
except ImportError:
    pyttsx3 = None


ROOT_DIR = os.path.dirname(
    os.path.dirname(
        os.path.abspath(__file__)
    )
)

if ROOT_DIR not in sys.path:
    sys.path.insert(0, ROOT_DIR)


from core.brain import JarvisBrain


class JarvisGUI:

    def __init__(self, root):

        self.root = root

        self.root.title("JARVIS")

        self.root.geometry("1200x760")

        self.root.minsize(900, 600)

        self.root.configure(
            bg="#03060b"
        )

        self.running = True
        self.processing = False
        self.speaking = False

        self.time = 0.0

        self.last_answer = ""

        self.particles = []

        self.brain = JarvisBrain()

        self.voice_engine = None

        self.create_voice()

        self.create_particles()

        self.create_interface()

        self.root.protocol(
            "WM_DELETE_WINDOW",
            self.close
        )

        self.animate()

    # ============================================
    # VOICE
    # ============================================

    def create_voice(self):

        if pyttsx3 is None:
            return

        try:

            self.voice_engine = pyttsx3.init(
                "sapi5"
            )

            self.voice_engine.setProperty(
                "rate",
                165
            )

            self.voice_engine.setProperty(
                "volume",
                1.0
            )

        except Exception as error:

            print(
                "VOICE INIT ERROR:",
                error
            )

            self.voice_engine = None

    def speak(self, text):

        if not self.voice_engine:
            return

        thread = threading.Thread(
            target=self.voice_thread,
            args=(text,)
        )

        thread.daemon = True

        thread.start()

    def voice_thread(self, text):

        try:

            self.root.after(
                0,
                self.start_speaking
            )

            self.voice_engine.say(text)

            self.voice_engine.runAndWait()

        except Exception as error:

            print(
                "VOICE ERROR:",
                error
            )

        finally:

            self.root.after(
                0,
                self.stop_speaking
            )

    def start_speaking(self):

        self.speaking = True

    def stop_speaking(self):

        self.speaking = False

    # ============================================
    # PARTICLES
    # ============================================

    def create_particles(self):

        self.particles = []

        for i in range(160):

            self.particles.append({
                "angle": random.random()
                * math.pi
                * 2,

                "radius": random.uniform(
                    160,
                    500
                ),

                "speed": random.uniform(
                    0.00012,
                    0.0005
                ),

                "phase": random.random()
                * math.pi
                * 2,

                "size": random.choice(
                    [1, 1, 1, 2]
                )
            })

    # ============================================
    # INTERFACE
    # ============================================

    def create_interface(self):

        self.canvas = tk.Canvas(
            self.root,
            bg="#03060b",
            highlightthickness=0
        )

        self.canvas.pack(
            fill="both",
            expand=True
        )

        self.input_frame = tk.Frame(
            self.root,
            bg="#080d16"
        )

        self.input_frame.place(
            relx=0.5,
            rely=0.91,
            anchor="center",
            relwidth=0.76,
            height=55
        )

        self.entry = tk.Entry(
            self.input_frame,
            bg="#0b121e",
            fg="#e7f2ff",
            insertbackground="#ffffff",
            relief="flat",
            font=("Arial", 13)
        )

        self.entry.pack(
            side="left",
            fill="both",
            expand=True,
            padx=12,
            pady=9
        )

        self.entry.bind(
            "<Return>",
            self.send_message
        )

        self.send_button = tk.Button(
            self.input_frame,
            text="SEND",
            command=self.send_message,
            bg="#17253a",
            fg="#e7f2ff",
            activebackground="#263d5c",
            activeforeground="#ffffff",
            relief="flat",
            bd=0,
            font=("Arial", 10, "bold")
        )

        self.send_button.pack(
            side="right",
            padx=8,
            pady=8
        )

        self.entry.focus_set()

    # ============================================
    # SEND MESSAGE
    # ============================================

    def send_message(self, event=None):

        if self.processing:
            return

        text = self.entry.get().strip()

        if not text:
            return

        self.entry.delete(
            0,
            tk.END
        )

        self.processing = True

        self.send_button.config(
            state="disabled"
        )

        thread = threading.Thread(
            target=self.process_message,
            args=(text,)
        )

        thread.daemon = True

        thread.start()

    # ============================================
    # AI
    # ============================================

    def process_message(self, text):

        try:

            answer = self.brain.ask(
                text
            )

        except Exception as error:

            answer = (
                "JARVIS ERROR: "
                + str(error)
            )

        self.root.after(
            0,
            lambda: self.show_answer(
                answer
            )
        )

    def show_answer(self, answer):

        self.last_answer = answer

        self.processing = False

        self.send_button.config(
            state="normal"
        )

        self.speak(
            answer
        )

    # ============================================
    # ANIMATION
    # ============================================

    def animate(self):

        if not self.running:
            return

        self.time += 0.022

        self.canvas.delete(
            "dynamic"
        )

        width = self.canvas.winfo_width()
        height = self.canvas.winfo_height()

        if width < 100:
            width = 1200

        if height < 100:
            height = 760

        cx = width * 0.5
        cy = height * 0.40

        # ----------------------------------------
        # TITLE
        # ----------------------------------------

        self.canvas.create_text(
            cx,
            42,
            text="J A R V I S",
            fill="#dcecff",
            font=("Arial", 24, "bold"),
            tags="dynamic"
        )

        if self.processing:

            status = "THINKING"

        elif self.speaking:

            status = "SPEAKING"

        else:

            status = "ONLINE"

        self.canvas.create_text(
            cx,
            70,
            text=status,
            fill="#718eac",
            font=("Arial", 9),
            tags="dynamic"
        )

        # ----------------------------------------
        # SPACE PARTICLES
        # ----------------------------------------

        for p in self.particles:

            p["angle"] += p["speed"]

            angle = p["angle"]

            radius = p["radius"]

            x = (
                cx
                + math.cos(angle)
                * radius
            )

            y = (
                cy
                + math.sin(angle)
                * radius
                * 0.45
            )

            drift = math.sin(
                self.time
                + p["phase"]
            ) * 2

            x += drift

            y += drift * 0.3

            size = p["size"]

            self.canvas.create_oval(
                x - size,
                y - size,
                x + size,
                y + size,
                fill="#7189a5",
                outline="",
                tags="dynamic"
            )

        # ----------------------------------------
        # CORE BREATHING
        # ----------------------------------------

        breathing = (
            math.sin(
                self.time * 1.2
            ) * 4
        )

        if self.processing:

            breathing += (
                math.sin(
                    self.time * 2.5
                ) * 3
            )

        if self.speaking:

            breathing += (
                math.sin(
                    self.time * 5
                ) * 7
            )

        core = 40 + breathing

        # ----------------------------------------
        # ELECTROMAGNETIC RINGS
        # ----------------------------------------

        for i in range(6):

            radius = (
                75
                + i * 31
            )

            wave = math.sin(
                self.time * 1.1
                + i * 0.7
            ) * 6

            if self.speaking:

                wave += math.sin(
                    self.time * 4
                    + i
                ) * 4

            rx = radius + wave

            ry = rx * 0.42

            self.canvas.create_oval(
                cx - rx,
                cy - ry,
                cx + rx,
                cy + ry,
                outline="#4c6985",
                width=1,
                tags="dynamic"
            )

        # ----------------------------------------
        # CENTRAL ENERGY WAVE
        # ----------------------------------------

        points = []

        wave_width = 280

        for i in range(121):

            x = (
                cx
                - wave_width
                + i
                * (
                    wave_width * 2
                    / 120
                )
            )

            normalized = (
                x - cx
            ) / wave_width

            envelope = (
                1
                - abs(normalized)
            )

            wave = math.sin(
                normalized * 11
                + self.time * 1.4
            )

            y = (
                cy
                + wave
                * 18
                * envelope
            )

            points.extend(
                [x, y]
            )

        self.canvas.create_line(
            *points,
            fill="#7997b6",
            width=1,
            smooth=True,
            tags="dynamic"
        )

        # ----------------------------------------
        # CORE
        # ----------------------------------------

        self.canvas.create_oval(
            cx - core,
            cy - core,
            cx + core,
            cy + core,
            outline="#d8eaff",
            width=2,
            tags="dynamic"
        )

        inner = core * 0.45

        self.canvas.create_oval(
            cx - inner,
            cy - inner,
            cx + inner,
            cy + inner,
            outline="#8ba9c8",
            width=2,
            tags="dynamic"
        )

        # ----------------------------------------
        # ORBITING ENERGY
        # ----------------------------------------

        for i in range(14):

            angle = (
                self.time * 0.35
                + i * math.pi / 7
            )

            radius = (
                112
                + math.sin(
                    self.time + i
                ) * 5
            )

            x = (
                cx
                + math.cos(angle)
                * radius
            )

            y = (
                cy
                + math.sin(angle)
                * radius
                * 0.40
            )

            self.canvas.create_oval(
                x - 2,
                y - 2,
                x + 2,
                y + 2,
                fill="#b8cee5",
                outline="",
                tags="dynamic"
            )

        # ----------------------------------------
        # RESPONSE
        # ----------------------------------------

        if self.last_answer:

            self.canvas.create_text(
                cx,
                height * 0.67,
                text=self.last_answer,
                fill="#dcecff",
                width=min(
                    width * 0.70,
                    850
                ),
                font=("Arial", 12),
                justify="center",
                tags="dynamic"
            )

        self.root.after(
            25,
            self.animate
        )

    # ============================================
    # CLOSE
    # ============================================

    def close(self):

        self.running = False

        try:

            if self.voice_engine:

                self.voice_engine.stop()

        except Exception:
            pass

        self.root.destroy()


def main():

    root = tk.Tk()

    JarvisGUI(
        root
    )

    root.mainloop()


if __name__ == "__main__":

    main()
